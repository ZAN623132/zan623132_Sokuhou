// ------------------------------------------------------------------- // アクセスカウンター処理（GitHub Pages / CORS完全対応版） // ------------------------------------------------------------------- async function updateVisitorCount() { const countEl = document.getElementById("visitor-count"); if (!countEl) return; try { // あなたのサイト専用の識別キー（英数字とハイフンのみ） const namespace = "meiji-meiji-rikujo-2026-sokuhou"; const key = "pv-count"; // CORS許可済みのカウンターAPI（カウントアップ: +1） const res = await fetch(`https://count.getloli.com/get/@${namespace}-${key}?theme=rule`); if (!res.ok) throw new Error(`HTTP Error: ${res.status}`); // SVG/JSONレスポンスから数値を抽出（またはCORS対応の代替API） // ※上記APIが重い場合のバックアップAPI（下記） const res2 = await fetch(`https://api.counterapi.dev/v2/${namespace}/${key}/up`); if (!res2.ok) throw new Error(`HTTP Error: ${res2.status}`); const data = await res2.json(); const count = data.data?.count ?? data.count; if (count !== undefined) { countEl.textContent = `| 累計アクセス: ${Number(count).toLocaleString()} 回`; } } catch (err) { console.error("アクセスカウンターエラー:", err); countEl.textContent = `| 累計アクセス: - 回`; } }
